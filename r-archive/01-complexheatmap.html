ComplexHeatmap setup

plot_mat <-
  plot_mat[row_order, ] %>%
  filter(row.names(.) %in% unique(kept_tfs))
dim(plot_mat)
head(plot_mat)

colSplit <- c(rep("active", 32), rep("repressive", 16))
mypal <- c(pal_npg("nrc", alpha = 0.8)(10), "#e9c46a", "#588157")
show_col(mypal)
names(mypal) <- unique(sapply(str_split(colnames(plot_mat), "-"), "[[", 1))

H3_labels <- split(1:48, str_split(colnames(plot_mat), "-", simplify = TRUE)[, 1])
names(H3_labels) <- str_replace_all(names(H3_labels), "H3", "")
names(H3_labels)[2] <- "5mC"

column_ha = HeatmapAnnotation(
  H3_mark = anno_block(
    align_to = H3_labels,
    panel_fun = function(index, nm) {
      grid.rect(gp = gpar(fill = "white"))
      grid.text(nm, 0.5, 0.5)
    }
  ),
  group = sapply(str_split(colnames(plot_mat), "-"), "[[", 2),
  col = list(
    group = c("C1" = "#4c72b0", "C2" = "#dd8452", "C3" = "#55a868", "C4" = "#c44e52"),
    histone_mark = mypal
  )
)

col_fun = colorRamp2(
  c(0.07, 0.5, 1, 3, 5, 7),
  c("#2166AC", "#4393C3", "#FFFFFF", "#F4A582", "#D6604D", "#B2182B")
)

h = Heatmap(
  as.matrix(plot_mat),
  name = "TF_frac+0.01/\nBG_frac+0.01",
  top_annotation = column_ha,
  cluster_rows = FALSE,
  cluster_columns = FALSE,
  show_column_names = FALSE,
  column_split = colSplit,
  col = col_fun
)
h
