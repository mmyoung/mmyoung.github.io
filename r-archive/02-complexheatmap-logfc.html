soc_FvsM_DEG_FC <- read.table("./soc_FvsM_DEG_FC_celltype_matrix.tsv", header = TRUE, sep = "\t", row.names = 1)
soc_FvsM_DEG_P <- read.table("./soc_FvsM_DEG_adjP_celltype_matrix.tsv", header = TRUE, sep = "\t", row.names = 1)

rm.neg <- plyr::colwise(function(x) {
  ifelse(x < 0.0001, "****",
    ifelse(x < 0.001, "***",
      ifelse(x < 0.01, "**",
        ifelse(x < 0.05, "*", ""))))
})
p_sign_df <- rm.neg(soc_FvsM_DEG_P)
p_sign_df[is.na(p_sign_df)] <- ""

library(circlize)
library(ComplexHeatmap)

col_fun = colorRamp2(c(-5, 0, 5), c("#0077b6", "white", "#e63946"))
p = Heatmap(
  soc_FvsM_DEG_FC,
  name = "soc F/M\navg_logFC",
  col = col_fun,
  cluster_columns = FALSE,
  show_row_dend = FALSE,
  row_names_gp = grid::gpar(fontsize = 4),
  cell_fun = function(j, i, x, y, width, height, fill) {
    gb = textGrob("*")
    gb_w = convertWidth(grobWidth(gb), "mm")
    gb_h = convertHeight(grobHeight(gb), "mm")
    grid.text(sprintf("%s", p_sign_df[i, j]), x, y - gb_h * 0.5 + gb_w * 0.4, gp = gpar(fontsize = 10))
  }
)
