library(data.table)
library(org.At.tair.db)
library(clusterProfiler)
library(tidyverse)
library(ComplexHeatmap)

for(cell_type in c("mesophyll", "mature-endodermis", "lignified-endodermis", "mature-xylem", "suberized-endodermis")) {
  TF_target_df <- read.csv(
    paste0("/Volumes/project/gzy8899/references/multiDAPseq_from_Sharon/", cell_type, "_enriched_tfs.target_gene_lists.csv"),
    header = TRUE
  )
  TF_target_df$tf_name <- ifelse((TF_target_df$tf_name == "") | is.na(TF_target_df$tf_name), TF_target_df$tf, TF_target_df$tf_name)
  TF_target_df$alias <- ifelse(is.na(TF_target_df$alias) | (TF_target_df$alias == ""), TF_target_df$gene, TF_target_df$alias)

  mat_plot <- reshape2::dcast(TF_target_df, alias ~ tf, value.var = "avg_log2FC") %>%
    column_to_rownames(var = "alias")
  mat_plot[is.na(mat_plot)] <- 0
  mat_plot <- mat_plot[apply(mat_plot, 1, sum) > 5, ]

  h = Heatmap(
    mat_plot,
    name = "IsTarget",
    cluster_rows = TRUE,
    cluster_columns = TRUE,
    row_km = 6,
    column_km = 7,
    row_names_gp = gpar(fontsize = 8),
    column_names_gp = gpar(fontsize = 8)
  )
  draw(h)
}
