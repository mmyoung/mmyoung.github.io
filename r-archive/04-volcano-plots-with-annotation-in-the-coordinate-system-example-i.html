edger_volcanoplot<- 
function(edger_res,title){
  p <-
    ggplot(data=edger_res$table, aes(x=logFC, y =-log10(PValue)))+
     geom_point(data=subset(edger_res$table,edger_res$table$PValue>0.05 | abs(edger_res$table$logFC)<1),color="#828C8C")+    
     geom_point(data=subset(edger_res$table,edger_res$table$PValue<0.05 & edger_res$table$logFC > 1),color="#bb3e03") +
     geom_point(data=subset(edger_res$table,edger_res$table$PValue<0.05 & edger_res$table$logFC < (-1)),color="#004fff") +
     geom_hline(yintercept = -log10(0.05),lty=4,lwd=0.6,alpha=0.8)+
     geom_vline(xintercept = c(-1,1),lty=4,lwd=0.6,alpha=0.8)+
     theme_bw()+
     theme(panel.border = element_blank(),
           panel.grid.major = element_blank(),
           panel.grid.minor = element_blank(),
           axis.line = element_line(colour = 'black'),
           axis.title.x = element_text(size = 12),
           axis.title.y = element_text(size = 12)     )+
     labs(x='log2FC',y='-log10 (Pvalue)',title=title)+
     theme(legend.position='none')
  up_gene_num <- 
    sum(edger_res$table$PValue<0.05 & edger_res$table$logFC > 1)
  down_gene_num <- 
    sum(edger_res$table$PValue<0.05 & edger_res$table$logFC < (-1))
  
  p <- 
  p+annotate("text",x=layer_scales(p)$x$range$range[2]*0.8,y=layer_scales(p)$y$range$range[2]*0.8,label=paste0("up: ",up_gene_num))+
    annotate("text",x=layer_scales(p)$x$range$range[1]*0.8,y=layer_scales(p)$y$range$range[2]*0.8,label=paste0("down: ",down_gene_num))+
    theme(plot.title = element_text(hjust = 0.5,vjust = 0.5))
  
    ggsave(filename=paste0(title,".pdf"),p,
           width = 4,height = 5)
}
