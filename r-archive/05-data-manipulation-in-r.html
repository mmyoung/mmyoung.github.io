Data manipulation in R

When doing correlation analysis for variables in a matrix, and want to make plots with Pvalue and R:

## an example
res2 <- combined_fc %>%
    select(log2FoldChange,tf,sample) %>%
    tidyr::spread(tf,log2FoldChange) %>%
    column_to_rownames(var="sample") %>%
    as.matrix() %>%
    rcorr()
    
## a function to organize the data
flattenCorrMatrix <- function(cormat, pmat) {
  ut <- upper.tri(cormat)
  data.frame(
    row = rownames(cormat)[row(cormat)[ut]],
    column = rownames(cormat)[col(cormat)[ut]],
    cor  =(cormat)[ut],
    p = pmat[ut]
    )
}

flattenCorrMatrix(res2$r, res2$P)

Filter out rows that show NA values in a dataframe:

combined_fc %>%
    select(log2FoldChange,tf,sample) %>%
    tidyr::spread(tf,log2FoldChange) %>%
    column_to_rownames(var="sample") %>%
    cor(use = "pairwise.complete.obs")%>%
    as.data.frame() %>%
    filter(if_any(everything(), is.na))
