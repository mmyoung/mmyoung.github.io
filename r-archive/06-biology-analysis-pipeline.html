import pandas as pd
import argparse

def identify_methylated_regions(df, window_df, methylation_threshold=0.25, cytosine_threshold=0.4):
    """
    Identify methylated regions in the genome, count cytosines using window sequences from a file.

    Parameters:
        df (DataFrame): DataFrame with columns ['chromosome', 'position', 'mC_cov', 'total_cov']
                        (assumes each row is a cytosine at single base resolution).
        window_df (DataFrame): DataFrame containing genome windows with columns ['chromosome', 'start', 'end', 'sequence'].
        methylation_threshold (float): Threshold for methylation level to consider a cytosine as methylated.
        cytosine_threshold (float): Threshold for the fraction of methylated cytosines to call a region methylated.

    Returns:
        methylated_regions (list): List of methylated regions, each entry is a tuple (chromosome, start, end, cytosine_count, methylated_fraction).
    """
    methylated_regions = []

    # Group by chromosome in the methylation data
    for chrom, group in df.groupby('chromosome'):
        # Sort by position
        group = group.sort_values('position')

        # Iterate over each row in the window DataFrame
        for _, row in window_df.iterrows():
            if row['chromosome'] == chrom:
                window_start = row['start']
                window_end = row['end']
                sequence = row['sequence']

                # Count the cytosines in the window sequence
                cytosine_count = sequence.upper().count('C')
                guanine_count = sequence.upper().count('G')

                # Select cytosines in the current window based on methylation data
                in_window = group[(group['position'] >= window_start) & (group['position'] < window_end)]

                if cytosine_count > 0 and len(in_window) > 0:
                    # Calculate fraction of cytosines with methylation level > methylation_threshold
                    sense_methylated_cytosines = in_window[in_window['mC_level']  > methylation_threshold & in_window['strand'] == "+"]
                    sense_methylated_fraction = len(sense_methylated_cytosines) / cytosine_count

                    # If more than cytosine_threshold of cytosines are methylated, call this region methylated
                    if sense_methylated_fraction > cytosine_threshold:
                        methylated_regions.append((chrom, window_start, window_end, "+", cytosine_count, sense_methylated_fraction, in_window['mC_level'].mean()))

                    anti_methylated_cytosines = in_window[in_window['mC_level'] > methylation_threshold & in_window['strand'] == "-"]
                    anti_methylated_fraction = len(anti_methylated_cytosines) / guanine_count

                    # If more than cytosine_threshold of cytosines are methylated, call this region methylated
                    if anti_methylated_fraction > cytosine_threshold:
                        methylated_regions.append((chrom, window_start, window_end, "-", guanine_count, anti_methylated_fraction, in_window['mC_level'].mean()))

    return methylated_regions

def main():
    # Set up argument parser
    parser = argparse.ArgumentParser(description='Identify methylated regions based on methylation data and genome windows.')
    
    parser.add_argument('--windows_file', required=True, help='Path to the genome windows file (CSV/TSV format).')
    parser.add_argument('--peaks_file', required=True, help='Path to the compressed peaks file (gzip compressed).')
    parser.add_argument('--output_file', required=True, help='Path to the output file to save methylated regions.')
    parser.add_argument('--methylation_threshold', type=float, default=0.25, help='Methylation level threshold (default: 0.25).')
    parser.add_argument('--cytosine_threshold', type=float, default=0.4, help='Cytosine methylation fraction threshold (default: 0.4).')
    
    args = parser.parse_args()

    # Read the genome window file
    window_columns = ['chromosome', 'start', 'end', 'sequence']
    window_df = pd.read_csv(args.windows_file, sep='\t', names=window_columns)  # Adjust separator if necessary

    # Read the compressed peaks file without header
    peak_columns = ['chromosome', 'position', 'strand', 'context', 'mC_cov', 'total_cov', 'sig']
    peaks_df = pd.read_csv(args.peaks_file, compression='gzip', sep='\t', names=peak_columns)
    peaks_df['mc_level'] = peaks_df['mC_cov'] / peaks_df['total_cov']

    # Identify methylated regions
    methylated_regions = identify_methylated_regions(peaks_df, window_df, methylation_threshold=args.methylation_threshold, cytosine_threshold=args.cytosine_threshold)

    # Convert the results into a DataFrame for easy saving
    methylated_regions_df = pd.DataFrame(methylated_regions, columns=['chromosome', 'start', 'end', 'cytosine_count', 'methylated_fraction'])

    # Save the methylated regions to a CSV file
    methylated_regions_df.to_csv(args.output_file, sep="\t", columns = ['chr',"start","end","strand","C_count","mC_frac","mC_level_avg"],index=False)

    print(f"Methylated regions saved to {args.output_file}")

if __name__ == '__main__':
    main()

For scaleRNA-seq, to extract barcode information from R1 reads, and correct them according to whitelist if the mismatch falls bellow the threshold. Detect ligation, pcr and RT barcode and UMI at the same time.

RT barcode: 8nt away from linker sequence

ligation: first 9nt of R1 reads

PCR: the whole length of I1 reads

UMIs: roughly the 8nt sequences after linker (can be corrected afterwards)

import argparse
import gzip
from Levenshtein import distance

# Function to parse barcode files
def read_barcodes(file):
    barcodes = []
    barcode_dict = {}
    with open(file, 'r') as fh:
        for line in fh:
            sequence, name = line.strip().split("\t")
            barcodes.append(sequence)
            barcode_dict[sequence] = name
    return barcodes, barcode_dict

# Function to open FASTQ files
def open_fastq(file):
    if file.endswith(".gz"):
        return gzip.open(file, 'rt')
    return open(file, 'r')

# Function to correct ligation barcode
def correct_lig_bc(sequence, lig_bc_list, start_index, sub_length, mismatch, lig_barcodes):
    distance_min = sub_length
    match_code = None
    match_idx = 0

    for barcode in lig_bc_list:
        subseq = sequence[start_index:start_index + sub_length]
        dis = distance(barcode, subseq)

        if dis < distance_min:
            distance_min = dis
            match_code = barcode
            match_idx = 1
            if dis == 0:
                break
        elif dis == distance_min:
            match_idx += 1

    if match_idx == 1 and distance_min == 0:
        return {
            "label": "exact",
            "lig_bc": match_code,
            "lig_bc_well": lig_barcodes.get(match_code, "unknown")
        }
    elif match_idx == 1 and distance_min <= mismatch:
        return {
            "label": "corrected",
            "lig_bc": match_code,
            "lig_bc_well": lig_barcodes.get(match_code, "unknown")
        }
    else:
        return {
            "label": "error",
            "lig_bc": match_code or "none",
            "lig_bc_well": lig_barcodes.get(match_code, "unknown")
        }

# Function to process RT barcode
def correct_rt_bc(sequence, rt_bc_list, sub_length, mismatch, rt_barcodes):
    distance_min = sub_length
    match_code = None
    match_idx = 0
    umi_seq = None

    start_index = sequence.find("TCAGAGC")
    if start_index == -1:
        return {"label": "error", "rt_bc": "none", "rt_bc_well": "unknown", "umi": "unknown"}

    for barcode in rt_bc_list:
        subseq = sequence[start_index + 15:start_index + 15 + sub_length]
        dis = distance(barcode, subseq)

        if dis < distance_min:
            distance_min = dis
            match_code = barcode
            match_idx = 1
            umi_seq = sequence[start_index + 7:start_index + 15]
            if dis == 0:
                break
        elif dis == distance_min:
            match_idx += 1

    if match_idx == 1 and distance_min == 0:
        return {
            "label": "exact",
            "rt_bc": match_code,
            "rt_bc_well": rt_barcodes.get(match_code, "unknown"),
            "umi": umi_seq
        }
    elif match_idx == 1 and distance_min <= mismatch:
        return {
            "label": "corrected",
            "rt_bc": match_code,
            "rt_bc_well": rt_barcodes.get(match_code, "unknown"),
            "umi": umi_seq
        }
    else:
        return {
            "label": "error",
            "rt_bc": match_code or "none",
            "rt_bc_well": rt_barcodes.get(match_code, "unknown"),
            "umi": umi_seq
        }

# Function to process PCR barcode
def correct_pcr_bc(sequence, pcr_bc_list, mismatch, pcr_barcodes):
    distance_min = len(sequence)
    match_code = None
    match_idx = 0

    for barcode in pcr_bc_list:
        dis = distance(barcode, sequence)
        if dis < distance_min:
            distance_min = dis
            match_code = barcode
            match_idx = 1
            if dis == 0:
                break
        elif dis == distance_min:
            match_idx += 1

    if match_idx == 1 and distance_min == 0:
        return {
            "label": "exact",
            "pcr_bc": match_code,
            "pcr_bc_well": pcr_barcodes.get(match_code, "unknown")
        }
    elif match_idx == 1 and distance_min <= mismatch:
        return {
            "label": "corrected",
            "pcr_bc": match_code,
            "pcr_bc_well": pcr_barcodes.get(match_code, "unknown")
        }
    else:
        return {
            "label": "error",
            "pcr_bc": match_code or "none",
            "pcr_bc_well": pcr_barcodes.get(match_code, "unknown")
        }

batch_size = 1000  # Adjust based on available memory and performance

def read_batch(file_handle, lines_per_record=4):
    """Reads a batch of FASTQ records."""
    batch = []
    for _ in range(batch_size):
        record = [file_handle.readline().strip() for _ in range(lines_per_record)]
        if len(record[0]) == 0:  # EOF reached
            break
        batch.append(record)
    return batch

# Main function
def main():
    parser = argparse.ArgumentParser(description="Calculate Levenshtein distances for barcodes")
    parser.add_argument("--lig_barcode", required=True)
    parser.add_argument("--rt_barcode", required=True)
    parser.add_argument("--pcr_barcode", required=True)
    parser.add_argument("--r1_fastq", required=True)
    parser.add_argument("--r2_fastq", required=True)
    parser.add_argument("--i1_fastq", required=True)
    parser.add_argument("--out_fastq", required=True)
    parser.add_argument("--barcode_sta_out", required=True)
    parser.add_argument("--barcode_well_out", required=True)

    args = parser.parse_args()

    lig_barcodes, lig_barcode_dict = read_barcodes(args.lig_barcode)
    rt_barcodes, rt_barcode_dict = read_barcodes(args.rt_barcode)
    pcr_barcodes, pcr_barcode_dict = read_barcodes(args.pcr_barcode)

    print(f"Loaded {len(lig_barcodes)} ligation barcodes")
    print(f"Loaded {len(rt_barcodes)} RT barcodes")
    print(f"Loaded {len(pcr_barcodes)} PCR barcodes")

    processed_fq_line = 0
    with open_fastq(args.r1_fastq) as r1, open_fastq(args.r2_fastq) as r2, open_fastq(args.i1_fastq) as i1, \
         gzip.open(args.out_fastq, 'wt') as out_fq, open(args.barcode_sta_out, 'w') as bc_sta, open(args.barcode_well_out, 'w') as bc_well:

        bc_well.write("ligation_well\trt_well\tpcr_well\n")
        ligation_bc_sta = {}
        rt_bc_sta = {}
        pcr_bc_sta = {}

        while True:
            r1_batch = read_batch(r1)
            i1_batch = read_batch(i1)
            r2_batch = read_batch(r2)

            if not r1_batch or not i1_batch or not r2_batch:
                break

            for r1_records, i1_records, r2_records in zip(r1_batch, i1_batch, r2_batch):
                # Process records as before

                processed_fq_line +=1

                lig_bc_match = correct_lig_bc(r1_records[1], lig_barcodes, 0, 9, 2, lig_barcode_dict) # mismatch 3
                rt_bc_match = correct_rt_bc(r1_records[1], rt_barcodes, 10, 2, rt_barcode_dict) # mismatch 3
                pcr_bc_match = correct_pcr_bc(i1_records[1], pcr_barcodes, 2, pcr_barcode_dict) # mismatch 3

                ligation_bc_sta[lig_bc_match["label"]] = ligation_bc_sta.get(lig_bc_match["label"], 0) + 1
                rt_bc_sta[rt_bc_match["label"]] = rt_bc_sta.get(rt_bc_match["label"], 0) + 1
                pcr_bc_sta[pcr_bc_match["label"]] = pcr_bc_sta.get(pcr_bc_match["label"], 0) + 1

                if lig_bc_match["label"] != "error" and rt_bc_match["label"] != "error" and pcr_bc_match["label"] != "error":
                    first_line = f"@{lig_bc_match['lig_bc']}{rt_bc_match['rt_bc']}{pcr_bc_match['pcr_bc']},{rt_bc_match['umi']},{r2_records[0][1:]}"
                    out_fq.write(f"{first_line}\n{r2_records[1]}\n{r2_records[2]}\n{r2_records[3]}\n")
                    bc_well.write(f"{lig_bc_match['lig_bc_well']}\t{rt_bc_match['rt_bc_well']}\t{pcr_bc_match['pcr_bc_well']}\n")

        for label, count in ligation_bc_sta.items():
            bc_sta.write(f"ligation\t{label}\t{count}\n")
        for label, count in rt_bc_sta.items():
            bc_sta.write(f"rt\t{label}\t{count}\n")
        for label, count in pcr_bc_sta.items():
            bc_sta.write(f"pcr\t{label}\t{count}\n")

    print("Processed record:", processed_fq_line)

if __name__ == "__main__":
    main()
