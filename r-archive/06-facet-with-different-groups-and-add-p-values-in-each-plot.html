library(tidyr)
library(dplyr)
summ <-
ratio_df %>%
filter(C_group=="C4",n_c4_peaks>40) %>%
group_by(histone_mark) %>%
summarize(R = cor.test(median, odd_ratio,method = "spearman",exact=FALSE)$estimate, p = cor.test(median, odd_ratio,method = "spearman",exact=FALSE)$p.value) %>%
mutate(R = formatC(R, format = "fg", digits = 3),
p = formatC(p, format = "g", digits = 3)) %>%
## to make a wide data to a long one
gather(key = measure, value = value, -histone_mark) %>%
unite("stat", measure, value, sep = " = ") %>%
## combine two columns together
group_by(histone_mark) %>%
summarise(both_stats = paste(stat, collapse = "\n"))## add the correaltion

(ratio_df %>%
filter(C_group=="C4",n_c4_peaks>40) %>%
ggplot(aes(x=median,y=odd_ratio))+
geom_point(aes(color=tf_color))+
facet_wrap(.~histone_mark,scale="free_y")+
scale_color_manual(values = c("group1"="#f28482","group2"="#f6bd60","group3"="#84a59d"))+
theme_minimal()+
geom_text(aes(x = -Inf, y = Inf, label = both_stats), data = summ,
hjust = -0.1, vjust = 1.1, size = 3)+
scale_x_continuous(name="distance to cds start")+
scale_y_continuous(name="proportion of DAPpeak overlap with markers/bg")+
theme(strip.text = element_text(size = 12))) %>%
ggsave(filename="/Volumes/project/gzy8899/lyang/Abrabidopsis/multiDAPseq_explore/analysis/tf_tss_distance_vs_enrichment_dotplot.pdf",width=10,height = 8)
