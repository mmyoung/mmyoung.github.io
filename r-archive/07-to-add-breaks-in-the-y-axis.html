To add breaks in the y-axis

library(scales)
trans <- function(x){pmin(x,100) + 0.01*pmax(x-100,0)}
yticks <- c(0, 500,1000, 2000, 4000, 10000, 15000, 20000)

umis_count$trans <- 
trans(umis_count$row)

umis_count %>% 
filter(sample=="AU" & umis>10) %>%
ggplot(aes(x=umis, y=trans)) + 
  geom_col()+
  geom_vline(aes(xintercept=1291),color="#e76f51",linetype="dashed")+
  scale_x_continuous(limits=c(0,20000))+
  scale_y_continuous(name="cell counts",expand=c(0,0),limits=c(0,NA), breaks=trans(yticks), labels=yticks)+
  theme_minimal()+
  theme(axis.text=element_text(size=12),axis.title=element_text(size=14))
