library(ggplot2)
library(reshape2)
library(Rmisc)
setwd("D:/课题和课题数据及代码/WXZ_YBX2/public_scRNA-seq")
TPM<-read.table("vital_gene.tpm",row.names=1,stringsAsFactors = F)
t_TPM <- as.data.frame(t(TPM))
for (i in colnames(t_TPM)[-1]){
  t_TPM[,i] <- as.numeric(as.character(t_TPM[,i]))
}
t_TPM$Sample<-factor(t_TPM$Sample,levels=rev(c("A1", "In", "BS", "BG2", "G1", "ePL", "mPL", "IPL", "L", "Z", "eP", "mP", "IP", "D", "MI", "MII", "RS2", "RS4", "RS6", "RS8")))
melted_TPM <- melt(t_TPM,id.vars=c("Sample"))
# 获取各个基因表达量的平均值
df_mean <- aggregate(melted_TPM$value, list(Group=melted_TPM$Sample,
                        gene=melted_TPM$variable), mean, na.rm=T)
# 获取各个基因表达量的标准差
df_sd <- aggregate(melted_TPM$value, list(Group=melted_TPM$Sample,
                        gene=melted_TPM$variable), sd, na.rm=T)
# 合并mean和sd
colnames(df_mean)[3] <- "mean"
colnames(df_sd)[3] <- "sd"
df_stat <- merge(df_mean, df_sd, by=c("Group", "gene"))
str(df_stat)
##接下来用df_stat统计的信息作图。
p<-ggplot(df_stat,aes(x=Group, y=mean, fill=gene))+
  geom_bar(stat="identity", position = "dodge")+
  geom_errorbar(aes(x=Group, ymin=mean-sd, ymax=mean+sd),color="black",width=.3) +
  scale_fill_brewer(palette = "Paired")+
  theme(axis.title.x=element_blank(), axis.text.x=element_blank(),axis.ticks.x=element_blank())+ #去掉横轴标题、刻度和刻度值
  theme(axis.title.y=element_blank())+ ###去掉y轴标题
  theme(axis.text.y = element_text(size = 25, face = "bold", angle = 0,family = "Times"))+
  theme(plot.title = element_text(size = 30,hjust = 0.5, face = "bold",family = "Times"))+ ##vjust,hjust这里一直调不好就用默认的吧
  theme(axis.ticks=element_line(colour="black",size=1,linetype=1,lineend=1),axis.ticks.length=unit(.4,"lines"))+ ##设置边框特征，刻度长度
  theme(legend.position = 0)+ ##去掉legend
  coord_flip() + ##将箱线图转置
  facet_grid(~gene,scale = "free")+ ##根据variable分面，固定坐标刻度一致
  theme(strip.text = element_text(colour = 'black', face = 'bold.italic', family="Times", size = rel(2.5)), strip.background = element_blank()) + ###strip.text设置分面标签的字体，颜色和大小，strip.background设置分面背景，这里去掉分面名上的灰色背景
  theme(panel.background = element_blank(),panel.border = element_rect(color="Black",fill=NA,size=2))
ggsave(p,file=paste("merge_barplot.pdf",sep=""),height=10,width=25)
ggsave(p,file=paste("merge_barplot.png",sep=""),height=10,width=25)
