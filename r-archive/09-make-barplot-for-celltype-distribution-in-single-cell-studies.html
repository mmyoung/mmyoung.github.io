options(repr.plot.width = 10, repr.plot.height = 4, repr.plot.res = 300)
seu_obj_recluster@meta.data %>%
  group_by(celltype,sampleID) %>%
  count() %>%
  group_by(sampleID) %>%
  mutate(percent=100*n/sum(n)) %>%
  ungroup() %>%
  mutate(celltype_color=case_when(celltype == "Sst-Inh" ~ "#C8C8C8", 
                                        celltype == "L5 IT" ~ "#FDDCFF",
                                        celltype == "L5 ET" ~ "#B28D9A",
                                        celltype == "L4 IT" ~ "#C77963",
                                        celltype == "CGE-Inh" ~ "#CEFF00",
                                        celltype == "L2/3 IT" ~ "#D9F0A3",
                                        celltype == "L6 CT" ~ "#0094C2",
                                        celltype == "ASC" ~ "#00B7FF",
                                        celltype == "Pvalb-Inh" ~ "#0640FF",
                                        celltype == "Micro" ~ "#21F4CC",
                                        celltype == "ODC" ~ "#0DFF00",
                                        celltype == "BG" ~ "#ADBF00"
                                       ))%>%
  ggplot(aes(x=sampleID,y=percent, fill=celltype_color)) +
  geom_col() +
  scale_fill_identity(guide = "legend", labels = unique(seu_obj_recluster@meta.data$celltype)) +
  ggtitle("Percentage of cell types per sample") +
  theme_minimal() +
  labs(x = "Sample ID", y = "Percentage", fill = "Cell Type")
