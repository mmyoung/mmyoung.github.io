Permutation test for peak overlaps

There’re always cases that we need to check the overrepresentation of a set of peak in another set of peaks, for instance, when I need to see if C4 DAP-seq peaks.

#!/bin/bash

#SBATCH --job-name=dmr_1
#SBATCH --output=dmr.out
#SBATCH --error=dmr.err
#SBATCH --time=36:00:00
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=2
#SBATCH --mem-per-cpu=8G
#SBATCH --partition=caslake
#SBATCH --account=pi-gzy8899

cd /project/gzy8899/lyang/tissue_specific_DNAme/analysis/new_MR_define/Ath_sdl_L_R
DAP_file=/project/gzy8899/references/multiDAPseq_from_Leo/N4_filtered-annotated-peaks_minfoldch5_minus-2000bp-to-plus-500bp_111623.tsv

simulation_peaks=/project/gzy8899/references/multiDAPseq_from_Leo/filtered-simulated_1ktfs_ath1ksites_minus-2000bp-to-plus-500bp.tsv

DMR_file=/project/gzy8899/lyang/tissue_specific_DNAme/analysis/new_MR_define/Ath_sdl_L_R_diff_regions.tsv

permutation_test_r=/project/gzy8899/lyang/tissue_specific_DNAme/analysis/new_MR_define/Ath_sdl_L_R/permutation_test.R

sed "1d" $DMR_file|awk '$4>1{print "Chr"$0}'|sortBed -i - > dmr_file.bed
cut -f 2-4 $simulation_peaks |sed "1d" > simulation_peak.bed

rm DAPpeak_DMR_ol_per.sta
touch DAPpeak_DMR_ol_per.sta
printf "tf\tC1_peak_num\tC2_peak_num\tC3_peak_num\tC4_peak_num\tC1_ol_peak_num\tC2_ol_peak_num\tC3_ol_peak_num\tC4_ol_peak_num\tC1_simulation_res\tC2_simulation_res\tC3_simulation_res\tC4_simulation_res\n" >> DAPpeak_DMR_ol_per.sta
for i in `cut -f 31 $DAP_file|sort|uniq`
do
    awk -v tf=${i} '$31==tf && $37=="1" && $26=="Arabidopsis_thaliana_Col-0"' $DAP_file|sed "1d"|cut -f 1-3|sortBed -i - >tmp_C1_peak.bed
    awk -v tf=${i} '$31==tf && $37=="2" && $26=="Arabidopsis_thaliana_Col-0"' $DAP_file|sed "1d"|cut -f 1-3|sortBed -i - >tmp_C2_peak.bed
    awk -v tf=${i} '$31==tf && $37=="3" && $26=="Arabidopsis_thaliana_Col-0"' $DAP_file|sed "1d"|cut -f 1-3|sortBed -i - >tmp_C3_peak.bed
    awk -v tf=${i} '$31==tf && $37=="4" && $26=="Arabidopsis_thaliana_Col-0"' $DAP_file|sed "1d"|cut -f 1-3|sortBed -i - >tmp_C4_peak.bed

    C1_peak=tmp_C1_peak.bed
    C2_peak=tmp_C2_peak.bed
    C3_peak=tmp_C3_peak.bed
    C4_peak=tmp_C4_peak.bed

    C1_peak_num=`cat $C1_peak|wc -l `
    C2_peak_num=`cat $C2_peak|wc -l`
    C3_peak_num=`cat $C3_peak|wc -l`
    C4_peak_num=`cat $C4_peak|wc -l`

    C1_simulation_res=$(Rscript $permutation_test_r simulation_peak.bed dmr_file.bed $C1_peak_num 1000)
    C2_simulation_res=$(Rscript $permutation_test_r simulation_peak.bed dmr_file.bed $C2_peak_num 1000)
    C3_simulation_res=$(Rscript $permutation_test_r simulation_peak.bed dmr_file.bed $C3_peak_num 1000)
    C4_simulation_res=$(Rscript $permutation_test_r simulation_peak.bed dmr_file.bed $C4_peak_num 1000)

    C1_ol_peak_num=`intersectBed -a $C1_peak -b dmr_file.bed|wc -l`
    C2_ol_peak_num=`intersectBed -a $C2_peak -b dmr_file.bed|wc -l`
    C3_ol_peak_num=`intersectBed -a $C3_peak -b dmr_file.bed|wc -l`
    C4_ol_peak_num=`intersectBed -a $C4_peak -b dmr_file.bed|wc -l`

    printf "%s\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%d\t%s\t%s\t%s\t%s\n" ${i} ${C1_peak_num} ${C2_peak_num} ${C3_peak_num} ${C4_peak_num} ${C1_ol_peak_num} ${C2_ol_peak_num} ${C3_ol_peak_num} ${C4_ol_peak_num} ${C1_simulation_res} ${C2_simulation_res} ${C3_simulation_res} ${C4_simulation_res}>> DAPpeak_DMR_ol_per.sta

    rm tmp*
done
#!/usr/bin/env Rscript
### permutation_test.R
# Run the R script and capture its output into a variable
## usage: overlap_counts=$(Rscript permutation_test.R simulation_peaks.txt dmr_peaks.txt 100 1000)

# Load necessary package
suppressPackageStartupMessages(library(GenomicRanges))

# Function to sample peaks, count overlaps, and repeat the process
sample_and_count_overlaps <- function(simulation_peaks, dmr_peaks, num_samples, num_repeats = 1000) {
  
  # Ensure inputs are GRanges objects
  if (!inherits(simulation_peaks, "GRanges") || !inherits(dmr_peaks, "GRanges")) {
    stop("Both simulation_peaks and dmr_peaks should be GRanges objects.")
  }

  # Use replicate to repeat the sampling and overlap counting process
  overlap_counts <- replicate(num_repeats, {
    # Randomly sample from the simulation peaks
    sampled_peaks <- simulation_peaks[sample(length(simulation_peaks), num_samples)]
    
    # Count overlaps directly
    length(findOverlaps(sampled_peaks, dmr_peaks))
  })
  
  # Concatenate counts into a single comma-separated string
  overlap_counts_str <- paste(overlap_counts, collapse = ",")
  
  return(overlap_counts_str)
}

# Function to load peaks from a file into a GRanges object
load_peaks <- function(file) {
  data <- read.table(file, header = FALSE)
  GRanges(seqnames = data$V1, ranges = IRanges(start = data$V2, end = data$V3))
}

# Main script to handle command-line arguments and execute the function
args <- commandArgs(trailingOnly = TRUE)
if (length(args) < 4) {
  stop("Usage: Rscript count_overlaps.R <simulation_peaks_file> <dmr_peaks_file> <num_samples> <num_repeats>")
}

# Parse command-line arguments
simulation_peaks_file <- args[1]
dmr_peaks_file <- args[2]
num_samples <- as.numeric(args[3])
num_repeats <- as.numeric(args[4])

# Load peak files
simulation_peaks <- load_peaks(simulation_peaks_file)
dmr_peaks <- load_peaks(dmr_peaks_file)

# Perform the sampling and counting process
result <- sample_and_count_overlaps(simulation_peaks, dmr_peaks, num_samples, num_repeats)

# Output the result
cat(result)
