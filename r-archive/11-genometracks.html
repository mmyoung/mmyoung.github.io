#!/usr/bin/perl
use strict;
use warnings;
use File::Temp qw(tempfile);
use IPC::Open2;

## This script is to split the mreged bam file into single cell files based on the CB field in the bam files

# Check for correct usage
if (@ARGV < 1) {
    die "Usage: perl split_bam_by_barcode.pl <input.bam>\n";
}

# Input BAM file
my $input_bam = $ARGV[0];

# Hash to store temporary files for each barcode
my %barcode_tempfiles;

# Open the input BAM file using samtools to read both header and data
open my $sam_in, "-|", "samtools view -h $input_bam" or die "Cannot open BAM file: $!";

# Process each line in the BAM file
while (my $line = <$sam_in>) {
    chomp $line;

    # If it's a header line, we need to save it to add to each output BAM file later
    if ($line =~ /^\@/) {
        foreach my $fh (values %barcode_tempfiles) {
            print $fh "$line\n";
        }
        next;
    }

    # Parse the cell barcode (CB tag) from the read line
    if ($line =~ /CB:Z:(\S+)/) {
        my $barcode = $1;

        # Create a new temporary file for each unique barcode, if not already present
        if (!exists $barcode_tempfiles{$barcode}) {
            my ($temp_fh, $temp_filename) = tempfile();
            $barcode_tempfiles{$barcode} = $temp_fh;
            print $temp_fh "$line\n";
        } else {
            # Write the read line to the respective barcode's temporary file
            my $temp_fh = $barcode_tempfiles{$barcode};
            print $temp_fh "$line\n";
        }
    }
}

# Close the input BAM file handle
close $sam_in;

# Write each temporary file to a BAM file with Samtools
foreach my $barcode (keys %barcode_tempfiles) {
    my $temp_fh = $barcode_tempfiles{$barcode};
    close $temp_fh;

    # Convert each temporary file to BAM and save it as <barcode>.bam
    my $pid = open2(my $out, my $in, "samtools view -Sb -o ${barcode}.bam -");
    open my $tempfile_in, "<", $temp_fh->filename or die "Cannot open temporary file: $!";
    while (my $line = <$tempfile_in>) {
        print $in $line;
    }
    close $tempfile_in;
    close $in;
    close $out;
}

print "BAM files split by cell barcode completed.\n";

1. To gather the single-cell .bam files to the same folder annotated as the same cell type (not necessarily move to same folder, create softlink works)

#!/bin/bash

# Check for correct usage
if [ "$#" -ne 2 ]; then
    echo "Usage: $0 <metadata_file> <bam_directory>"
    echo "Example: $0 metadata.tsv /path/to/bam_files"
    exit 1
fi

# Assign command-line arguments to variables
metadata_file=$1     # Path to metadata file
bam_directory=$2     # Path to directory containing BAM files

cd /project/zhuzhuzhang/lyang/scaleBio_RNA_rat_brain/out_mRatBN/make_pseudobulk_bigwig/sample_celltype_gather_bam

# Loop through each line in the metadata file
while IFS=$'\t' read -r cell_id sample_name cell_type; do
    # Skip the header row (assuming 'CellID', 'SampleName', 'CellType' as column headers)
    if [[ "$cell_id" == "CellID" ]]; then
        continue
    fi

    # Define the directory path for each cell type in each sample
    target_directory="${sample_name}/${cell_type}"

    # Create the target directory if it doesn't exist
    mkdir -p "$target_directory"

    # Define the source BAM file and target soft link path
    source_bam="${bam_directory}/${cell_id}.bam"
    target_link="${target_directory}/${cell_id}.bam"

    # Check if the BAM file exists before creating the link
    if [ -f "$source_bam" ]; then
        ln -s "$source_bam" "$target_link"
        echo "Created soft link for $cell_id in $target_directory"
    else
        echo "Warning: BAM file for $cell_id not found in $bam_directory"
    fi

done < "$metadata_file"
