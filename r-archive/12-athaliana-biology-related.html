plot the peak distribution around ATG

library(GenomicFeatures)
library(ChIPseeker)
library(dplyr)
library(ggplot2)
library(tidyr)

cds_primary_bed <- read.table("/Volumes/project/gzy8899/references/multiDAPseq_from_Leo/n4_n10_narrowpeaks/genome_annotations/Arabidopsis_thaliana_Col-0_cds_primary.gff",header = F,stringsAsFactors = F) %>%
makeGRangesFromDataFrame(seqnames.field = "V1",start.field = "V4",end.field = "V5",strand.field = "V7")

ATG_gr <- cds_primary_bed
ERF_targeted_ERF_peaks %>%
makeGRangesFromDataFrame(seqnames.field = "peak_chr",
start.field = "peak_start",
end.field = "peak_end") -> ERF_targeted_ERF_peaks_GR

tagMatrix  <- getTagMatrix(ERF_targeted_ERF_peaks_GR,
windows = makeBioRegionFromGranges(ATG_gr,type = "start_site",upstream = 2000, downstream = 500,by="gene"))
plotAvgProf(tagMatrix, xlim=c(-2000, 500),
xlab="Genomic Region (5'->3')", ylab = "Peak Count Frequency",origin_label = "ATG")

Athali_txdb<- makeTxDbFromGFF("/Volumes/project/gzy8899/references/multiDAPseq_from_Leo/n4_n10_narrowpeaks/genome_annotations/Arabidopsis_thaliana_Col-0.gff")
promoter <- getPromoters(TxDb=Athali_txdb, upstream=2000, downstream=500)
ERF_targeted_ERF_peaks %>%
makeGRangesFromDataFrame(seqnames.field = "peak_chr",
start.field = "peak_start",
end.field = "peak_end") %>%
getTagMatrix(windows = promoter) ->tagMatrix

plotAvgProf(tagMatrix, xlim=c(-2000, 500),
xlab="Genomic Region (5'->3')", ylab = "Peak Count Frequency")
