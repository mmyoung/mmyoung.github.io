To make the sequencing results into html report and embed the motif logos into the file:

library(dplyr)
library(DT)
library(kableExtra)
library(base64enc)

# Load the data
sample_data <- read.table("/Volumes/lyang/20250307_DAP-RNA_pooled_sequencing/src/Crubella.res.summary",sep="\t")
colnames(sample_data) <- c("sample","raw reads","mapped reads","peak#","FRiP_score")

# Define the image directory
image_dir <- "/Volumes/lyang/20250307_DAP-RNA_pooled_sequencing/Crubella_pipeline_out/meme_output"

head(sample_data)

# Add image paths to the data frame
sample_data <- sample_data %>%
mutate(Image_Path = paste0(image_dir, "/", sample, "_meme/logo1.png")) %>%
mutate(FRiP_score = round(100* FRiP_score,2))

sample_data$Image <- sapply(sample_data$Image_Path, function(img) {
if (file.exists(img)) {
encoded <- base64encode(img)
ext <- tools::file_ext(img)
sprintf('<img src="data:image/%s;base64,%s" style="max-width:100px;">', ext, encoded)
} else {
"Image not found"
}
})

sample_data <- sample_data %>% select(-Image_Path)

# Generate HTML table with embedded images
kable(sample_data, escape = FALSE,format = "html") %>%
kable_styling("striped", full_width = FALSE)
rmarkdown::render("report.Rmd")
