ggplot based heatmap (exhibiting GO terms from either promoter or distal targets of the same TF)

the trick is to use facet to split the TFs and plot only right borders for each panel to split groups

start with data like this:

data %>% 
group_by(tf_name) %>%
mutate(group_n=n()) %>%
filter(group_n==2) %>%
arrange(tf_name,group) %>%
    mutate(
        Description = factor(Description, levels = unique(Description)),
        # Create combined TF-group identifier
        tf_group = paste(tf_name, group, sep = "_"),
        # Order TFs so distal and promoter are adjacent
        tf_group = factor(tf_group, levels = unique(tf_group[order(tf_name, group)]))
    ) %>%
    ggplot(aes(x = tf_group, y = Description)) +
    geom_tile(aes(fill = p.adjust, color = group), size = 1,linewidth = 0.5) +
    scale_fill_gradient(low = "red", high = "white", trans = "log10", name = "p.adjust") +
    scale_color_manual(values = c("distal" = "#3a5a40", "promoter" = "#a5a58d"), 
                      name = "Target Type",
                      guide = guide_legend(
                    override.aes = list(
                        fill = "white",  # Set fill to white or transparent
                        color = c("#3a5a40", "#a5a58d"),
                        size = 1.5  # Make border more visible
            ))) +
    facet_wrap(.~tf_name,nrow = 1,scales = "free_x",strip.position = "bottom")+
    theme_minimal() +
    theme(
    panel.border = element_blank(),
    panel.background = element_blank(),
    # Add only right border to each panel
    panel.grid = element_blank(),  # Remove grid if desired

  ) +

plot exhibiting the number of DEGs in each time point in each cell type

start with DEG analysis results from DESeq2, with extra column denoting time points and celltype. The tricky part is to impute the 0 values in the matrix, because otherwise there will be no square plotted in an empty cell.

p1 <- time_course_DEG_pseudo %>%
    filter(celltype %in% !!celltype_label) %>%
  # Create a complete grid of all celltype and age combinations first
  {
    # Get all unique celltypes and ages
    all_celltypes <- unique(.$celltype)
    all_ages <- c("P30","P50", "P65", "P80", "P95", "P120")
    
    # Filter your data
    filtered_data <- filter(., p_val_adj < 0.05, avg_log2FC > 1) %>%
      group_by(celltype, age) %>%
      summarise(count = n()) %>%
      ungroup()
    
    # Create complete grid with all combinations
    complete_grid <- expand.grid(
      celltype = all_celltypes,
      age = all_ages,
      stringsAsFactors = FALSE
    )
    
    # Join with your filtered data
    left_join(complete_grid, filtered_data, by = c("celltype", "age")) %>%
      # Replace NA with 0
      mutate(count = replace_na(count, 0))
  } %>%
  mutate(celltype = factor(celltype, levels = rev(celltype_label))) %>%
  mutate(age = factor(age, levels = c("P30","P50", "P65", "P80", "P95", "P120"))) %>%
  ggplot(aes(x = age, y = celltype, fill = count)) +
  geom_tile(color = "black") +
  geom_text(aes(label = count), color = "#495057", size = 3) +
  labs(x = "", y = "", title = "p_val_adj<0.05,avg_log2FC>1") +
  scale_fill_gradient2(low = "white", high = "red") +
  scale_x_discrete(expand = c(0,0)) +
  scale_y_discrete(expand = c(0,0)) +
  theme_linedraw() +
  theme(legend.position = "none",
        plot.title = element_text(size = 7),
        panel.grid.major = element_blank())
