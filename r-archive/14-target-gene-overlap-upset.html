target_gene_ls = list(C1=gene_Csites_count$Target[gene_Csites_count$n1>0],
    C2=gene_Csites_count$Target[gene_Csites_count$n2>0],
    C3=gene_Csites_count$Target[gene_Csites_count$n3>0],
    C4=gene_Csites_count$Target[gene_Csites_count$n4>0])

## plot for foldchange of 0.3
m = make_comb_mat(target_gene_ls)
cs = comb_size(m)
options(repr.plot.width = 18, repr.plot.height = 5, repr.plot.res = 300)
## only plot the sets that have more than 10 elements
ht = UpSet(m,set_order = c(1,2,3,4),
        comb_order = order(cs,decreasing = TRUE),
        top_annotation = HeatmapAnnotation(
        "Genre Intersections" = anno_barplot(cs, 
            ylim = c(0, max(cs)*1.1),
            border = FALSE, 
            gp = gpar(fill = "black"), 
            height = unit(4, "cm")
        )))
options(repr.plot.width = 8, repr.plot.height = 4)
ht = draw(ht)
od = column_order(ht)
decorate_annotation("Genre Intersections", {
    grid.text(cs[od], x = seq_along(cs), y = unit(cs[od], "native") + unit(2, "pt"), 
        default.units = "native", just = c("left", "bottom"), 
        gp = gpar(fontsize = 8, col = "#404040"), rot = 30)
})
