#!/bin/bash

cd /project/zhuzhuzhang/lyang/scaleBio_RNA_rat_brain/out_mRatBN/make_pseudobulk_bigwig
awk -F "\t" 'BEGIN{OFS="\t"}{print $8,$9,$10,$7,$6,$11,$2}' /project/zhuzhuzhang/lyang/scaleBio_RNA_rat_brain/out_mRatBN/isoVSsoc_DEG_total_list.txt|sed "1d" |sed s/\\//""/g|sed s/" "/""/g >iso_soc_DEG_bed_for_pygenometrack.bed

while read chr start end gene celltype strand fc ;do
    pyGenomeTracks --tracks ./iso_soc_DEG_normalized_track_plot/${celltype}_track.ini --region $chr:$start-$(($end+1)) -o ./iso_soc_DEG_normalized_track_plot/plots/${celltype}_${gene}.png --title ${celltype}" isoVSsoc logFC: "${fc}
done < /project/zhuzhuzhang/lyang/scaleBio_RNA_rat_brain/out_mRatBN/make_pseudobulk_bigwig/iso_soc_DEG_bed_for_pygenometrack.bed
