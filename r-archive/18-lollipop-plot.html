# Library
library(tidyverse)
# Create data
value1=abs(rnorm(26))*2
data=data.frame(x=LETTERS[1:26], value1=value1, value2=value1+1+rnorm(26, sd=1) )
# Reorder data using average?
data = data %>% rowwise() %>% mutate( mymean = mean(c(value1,value2) )) %>% arrange(mymean) %>% mutate(x=factor(x, x))
# plot
ggplot(data) +
  geom_segment( aes(x=x, xend=x, y=value1, yend=value2), color="grey") +
  geom_point( aes(x=x, y=value1), color=rgb(0.2,0.7,0.1,0.5), size=3 ) +
  geom_point( aes(x=x, y=value2), color=rgb(0.7,0.2,0.1,0.5), size=3 ) +
  coord_flip()
# With a bit more style
ggplot(data) +
  geom_segment( aes(x=x, xend=x, y=value1, yend=value2), color="grey") +
  geom_point( aes(x=x, y=value1), color=rgb(0.2,0.7,0.1,0.5), size=3 ) +
  geom_point( aes(x=x, y=value2), color=rgb(0.7,0.2,0.1,0.5), size=3 ) +
  coord_flip()+
  theme_light() +
  theme(
    legend.position = "none",
    panel.border = element_blank(),
  ) +
  xlab("") +
  ylab("Value of Y")
library(reshape2)
CIMP_num_in_sig<-
  data.frame(unclass(table(Full_CGI_anno$signal_group,Full_CGI_anno$group)))
CIMP_num_in_sig$CGIs <- rowSums(CIMP_num_in_sig)
lollipop_data<- apply(CIMP_num_in_sig,2,function(x){x/sum(x)}) %>%
  as.data.frame() %>%
  mutate(sig.group=row.names(.)) %>%
  select(sig.group,A_CIMP,IDH_CIMP,CGIs) %>%
  melt() ## make the data 
lollipop_data$x <- c(seq(1,5),seq(1,5)+0.1, seq(1,5)+0.2) ## add x axis postions manually 
lollipop_data
p<-
  ggplot(lollipop_data,aes(x=x,y=value))+
  geom_segment( aes(x=x, xend=x, y=0, yend=value), color="skyblue")+
  geom_point( aes(color=variable), size=3, alpha=0.8) +
  scale_y_continuous(limits = c(0,1),name = "Ratio",expand = c(0,0))+
  scale_color_manual(name="group",
                     values = list("CGIs"="#7d8597","A_CIMP"="#4755B8","IDH_CIMP"="#FFCE33"))+
  scale_x_discrete(limits=c(1.1,2.1,3.1,4.1,5.1),labels=c("multi","sig.1","sig.2","sig.3","sig.4"))+
  theme_bw() +
  theme(
    panel.grid.major.x = element_blank(),
    panel.border = element_blank(),
    axis.ticks.x = element_blank(),
    axis.text = element_text(size = 12,colour = "black"),
    axis.title.y = element_text(size=12),
    plot.margin = unit(c(1,1,1,1), "cm")
  ) +
