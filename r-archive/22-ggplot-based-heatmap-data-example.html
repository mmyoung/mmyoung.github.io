data %>% 
group_by(tf_name) %>%
mutate(group_n=n()) %>%
filter(group_n==2) %>%
arrange(tf_name,group) %>%
    mutate(
        Description = factor(Description, levels = unique(Description)),
        # Create combined TF-group identifier
        tf_group = paste(tf_name, group, sep = "_"),
        # Order TFs so distal and promoter are adjacent
        tf_group = factor(tf_group, levels = unique(tf_group[order(tf_name, group)]))
    ) %>%
    ggplot(aes(x = tf_group, y = Description)) +
    geom_tile(aes(fill = p.adjust, color = group), size = 1,linewidth = 0.5) +
    scale_fill_gradient(low = "red", high = "white", trans = "log10", name = "p.adjust") +
    scale_color_manual(values = c("distal" = "#3a5a40", "promoter" = "#a5a58d"), 
                      name = "Target Type",
                      guide = guide_legend(
                    override.aes = list(
                        fill = "white",  # Set fill to white or transparent
                        color = c("#3a5a40", "#a5a58d"),
                        size = 1.5  # Make border more visible
            ))) +
    facet_wrap(.~tf_name,nrow = 1,scales = "free_x",strip.position = "bottom")+
    theme_minimal() +
    theme(
    panel.border = element_blank(),
    panel.background = element_blank(),
    # Add only right border to each panel
    panel.grid = element_blank(),  # Remove grid if desired
    strip.background = element_blank()
  ) +
  theme(
        # Make x-axis text invisible but keep spacing
        axis.text.x = element_text(color = "transparent"),
        axis.ticks.x = element_blank(),
        axis.title.x = element_blank(),
        axis.text.y = element_text(size = 10,color="black"),
        # Strip styling
        strip.text = element_text(size = 10,angle=90,hjust = 1)
    )+
  # Manually add right borders using geom_vline at panel edges
  geom_vline(xintercept = Inf, color = "black", linetype="dashed",size = 0.5)+
    labs(x = "Transcription Factor (Target Type)", y = "GO Terms"
