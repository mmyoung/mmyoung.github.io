cell_type <- "mesophyll"
TF_target_df <- read.csv(
  paste0("/Volumes/project/gzy8899/references/multiDAPseq_from_Sharon/", cell_type, "_enriched_tfs.target_gene_lists.csv"),
  header = TRUE
)
TF_target_df$tf_name <- ifelse((TF_target_df$tf_name == "") | is.na(TF_target_df$tf_name), TF_target_df$tf, TF_target_df$tf_name)
TF_target_df$alias <- ifelse(is.na(TF_target_df$alias) | (TF_target_df$alias == ""), TF_target_df$gene, TF_target_df$alias)

mat_plot <- reshape2::dcast(TF_target_df, gene ~ tf, value.var = "avg_log2FC") %>%
  column_to_rownames(var = "gene")
mat_plot[is.na(mat_plot)] <- 0

set.seed(0034)
h = Heatmap(
  mat_plot,
  name = "IsTarget",
  cluster_rows = TRUE,
  cluster_columns = TRUE,
  row_km = 6,
  column_km = 7,
  row_names_gp = gpar(fontsize = 8),
  column_names_gp = gpar(fontsize = 8)
)
draw_h = draw(h)
rcl.list = row_order(draw_h)

GO_ls <- lapply(rcl.list, function(x) {
  enrichGO(row.names(mat_plot)[x], OrgDb = org.At.tair.db, keyType = "TAIR", ont = "ALL")@result
})

text_box <- do.call(rbind, lapply(GO_ls, function(x) {
  x[x$pvalue < 0.05, c("Description", "geneID")]
})) %>%
  separate_rows(geneID) %>%
  pivot_wider(names_from = "Description", values_from = "Description", values_fill = 0, values_fn = function(x) 1) %>%
  column_to_rownames(var = "geneID")

h1 = Heatmap(mat_plot[unlist(rcl.list), ], name = "avg_log2FC")
h2 = Heatmap(text_box[rownames(mat_plot), ], show_row_names = FALSE, show_heatmap_legend = FALSE)
h1 + h2
